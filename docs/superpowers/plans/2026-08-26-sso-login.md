# SSO 单点登录（UniSSO）实施计划

日期：2026-08-26
规格：[docs/superpowers/specs/2026-08-26-sso-login-design.md](../specs/2026-08-26-sso-login-design.md)（含 S1 契约实测与生产 e2e 结论）

## 目标

集市接入 UniSSO 单点登录：登录页新增 SSO 入口，SSO 回调后按绑定关系自动登录或引导绑定；
已登录用户可在设置页发起绑定/解绑。原有邮箱密码登录、鉴权中间件、令牌体系**零改动**。

## 成功标准

1. `go build .` 与 `go test ./service/ ./controller/ ./repository/gorm/ ./config/ ./util/` 全绿（仓库根目录执行，不用 `./...`，历史死目录会编译失败）。
2. 本地 curl 走查：`/api/sso/login` 302 到 authorize（含 state/PKCE）；伪造 state 的 callback 302 到 `error=state`；伪造 ticket 的 exchange/bind 401。
3. 回归：邮箱登录、refresh、PAT 现有用例不改动即通过。
4. 前端 `npm run build`（`newSSE/`）通过；登录页有 SSO 按钮；结果页/绑定页路由可达。

## 关键事实（已实测确认）

- 生产 UniSSO：`https://sso.ssemarket.cn`（v1.1.0），authorize/token/userinfo 端点全链路 e2e 通过（见规格 8.1.2）。
- client_id：`client_uKTL00HcD4VAXyKI2TkBzA`；client_secret：`TvPaKBlMtbQvOUCeHINOCK8meA6ttW5VpSkLPRIkwwk`（只进后端环境变量，禁止入 git/前端）。
- redirect_uri：`https://ssemarket.cn/api/sso/callback`。
- scope 用 `openid profile email`（profile 自动展开为细粒度 profile:*）。
- token 端点接受 `client_secret` 直接 POST 表单；userinfo 需 `Authorization: Bearer <access_token>`；access_token 1 小时。
- 集市库：MySQL `ssemarket` 库（生产实例 ≠ 远程调试库 175.178.90.215:3506——**建表必须在生产实例上执行**，用 docker exec 进生产 db 容器）。
- 前后端同域部署（`/api` 反代）；密码传输走 `util.Encrypt/Decrypt` AES（`16bit secret key`）。
- Redis：`core.MyRedis`（go-redis v9，支持 `GetDel`）。
- 回调是浏览器顶层导航（302 链），不受 CORS 约束；ticket 走 POST body + Authorization，无 token-in-URL。

## 安全设计要点（实现时不可偏离）

- `state` 与 `ticket` 均一次性（Redis `GETDEL`），state TTL 10min，ticket TTL 2min。
- PKCE S256：verifier 64 字节随机 base64url；`challenge = BASE64URL(SHA256(verifier))`（RFC 7636）。
- ticket 的 login/bound 分支一次性消费（并发第二次 GETDEL 落空 → 401）；bind_needed 分支可重复 peek（等待绑定动作消费）。
- `bind/start` 需登录态（StrictAuth）；`bind_uid` 与回调用户不一致或双向均已绑 → `error=conflict`。
- client_secret 只从 `config.SsoClientSecret()` 读取，不落日志、不进任何响应。

```go
package gormrepo

import (
	"context"
	"errors"

	"loginTest/domain"
	"loginTest/model"
	"loginTest/repository"

	"gorm.io/gorm"
)

type ssoBindingRepository struct {
	db *gorm.DB
}

// NewSsoBindingRepository UniSSO 绑定关系仓储。
func NewSsoBindingRepository(db *gorm.DB) repository.SsoBindingRepository {
	return &ssoBindingRepository{db: db}
}

func (r *ssoBindingRepository) FindBySub(ctx context.Context, sub string) (*domain.SsoBinding, error) {
	var row model.SsoBinding
	err := r.db.WithContext(ctx).Where("sso_sub = ?", sub).First(&row).Error
	return toDomainSsoBinding(row, err)
}

func (r *ssoBindingRepository) FindByUserID(ctx context.Context, userID int) (*domain.SsoBinding, error) {
	var row model.SsoBinding
	err := r.db.WithContext(ctx).Where("local_user_id = ?", userID).First(&row).Error
	return toDomainSsoBinding(row, err)
}

func (r *ssoBindingRepository) Create(ctx context.Context, b *domain.SsoBinding) error {
	row := model.SsoBinding{
		LocalUserID: b.LocalUserID,
		SsoSub:      b.SsoSub,
		SsoUsername: b.SsoUsername,
		CreatedAt:   b.CreatedAt,
	}
	if err := r.db.WithContext(ctx).Create(&row).Error; err != nil {
		if isDuplicateEntryError(err) {
			return domain.ErrConflict
		}
		return err
	}
	b.ID = row.ID
	return nil
}

func (r *ssoBindingRepository) DeleteByUserID(ctx context.Context, userID int) error {
	res := r.db.WithContext(ctx).
		Where("local_user_id = ?", userID).Delete(&model.SsoBinding{})
	if res.Error != nil {
		return res.Error
	}
	if res.RowsAffected == 0 {
		return domain.ErrNotFound
	}
	return nil
}

func toDomainSsoBinding(row model.SsoBinding, err error) (*domain.SsoBinding, error) {
	if err != nil {
		if errors.Is(err, gorm.ErrRecordNotFound) {
			return nil, domain.ErrNotFound
		}
		return nil, err
	}
	return &domain.SsoBinding{
		ID:          row.ID,
		LocalUserID: row.LocalUserID,
		SsoSub:      row.SsoSub,
		SsoUsername: row.SsoUsername,
		CreatedAt:   row.CreatedAt,
	}, nil
}
```

## Task 4 — SsoService 实现（S3 核心）

### 4.1 `service/interfaces.go` 追加接口

```go
// SsoService UniSSO 单点登录服务。
type SsoService interface {
	// BuildAuthorizeURL 生成 state+PKCE 存 Redis，返回 UniSSO authorize 地址。
	// bindUID>0 表示已登录用户主动绑定；SSO 未启用返回 ErrBadRequest。
	BuildAuthorizeURL(ctx context.Context, bindUID int) (string, error)
	// ConsumeCallback 消费回调：校验 state（GETDEL 一次性）→ 换 token → 取 userinfo
	// → 按绑定关系分派一次性 ticket。错误用 ErrSsoStateInvalid/ErrSsoExchange/ErrSsoConflict 区分。
	ConsumeCallback(ctx context.Context, code, state string) (string, error)
	// ExchangeTicket ticket → 登录令牌（login/bound，一次性）或 bind_needed 信息（可重复读）。
	ExchangeTicket(ctx context.Context, ticket string) (*domain.SsoExchangeResult, error)
	// BindWithCredentials bind_needed ticket + 集市邮箱密码 → 写绑定 + 发令牌（ticket 一次性消费）。
	BindWithCredentials(ctx context.Context, ticket, email, password string) (*domain.AuthTokens, error)
	// GetStatus 查询用户绑定状态（未绑定返回 Bound=false，不报错）。
	GetStatus(ctx context.Context, userID int) (*domain.SsoStatus, error)
	// Unbind 验证密码后删除绑定。
	Unbind(ctx context.Context, userID int, password string) error
}
```

```go
package service

import (
	"context"
	"crypto/rand"
	"crypto/sha256"
	"encoding/base64"
	"encoding/json"
	"errors"
	"fmt"
	"io"
	"net/http"
	"net/url"
	"strings"
	"time"

	"loginTest/config"
	"loginTest/core"
	"loginTest/domain"
	"loginTest/repository"
	"loginTest/util"

	"golang.org/x/crypto/bcrypt"
)

const (
	ssoStateTTL  = 10 * time.Minute
	ssoTicketTTL = 2 * time.Minute
	ssoScope     = "openid profile email"

	// ticket 类型
	ssoTicketLogin      = "login"
	ssoTicketBindNeeded = "bind_needed"
	ssoTicketBound      = "bound"
)

// 回调阶段错误（controller 映射为 302 ?error=state|exchange|conflict）。
var (
	ErrSsoStateInvalid = errors.New("sso state invalid or replayed")
	ErrSsoExchange     = errors.New("sso token exchange failed")
	ErrSsoConflict     = errors.New("sso binding conflict")
)

// ssoKV state/ticket 的一次性 KV；生产为 Redis，单测注入内存实现。
type ssoKV interface {
	Set(ctx context.Context, key, val string, ttl time.Duration) error
	Get(ctx context.Context, key string) (string, error)
	GetDel(ctx context.Context, key string) (string, error)
}

type redisSsoKV struct{}

func (redisSsoKV) Set(ctx context.Context, key, val string, ttl time.Duration) error {
	return core.MyRedis.Set(ctx, key, val, ttl).Err()
}

func (redisSsoKV) Get(ctx context.Context, key string) (string, error) {
	return core.MyRedis.Get(ctx, key).Result()
}

func (redisSsoKV) GetDel(ctx context.Context, key string) (string, error) {
	return core.MyRedis.GetDel(ctx, key).Result()
}

// SsoUserAuthRepo 收窄依赖：绑定/解绑验证只需要这两个查询（gorm.UserRepository 已满足）。
type SsoUserAuthRepo interface {
	FindByID(ctx context.Context, id int) (*domain.User, error)
	FindAuthByEmail(ctx context.Context, email string) (*domain.User, string, error)
}

type ssoService struct {
	bindings repository.SsoBindingRepository
	users    SsoUserAuthRepo
	tokens   UserService
	kv       ssoKV
	http     *http.Client
}

// NewSsoService 生产装配：Redis KV + 真实 HTTP。
func NewSsoService(bindings repository.SsoBindingRepository, users SsoUserAuthRepo, tokens UserService) SsoService {
	return &ssoService{
		bindings: bindings,
		users:    users,
		tokens:   tokens,
		kv:       redisSsoKV{},
		http:     &http.Client{Timeout: 10 * time.Second},
	}
}

// SsoPKCEVerifier 生成高熵 code_verifier（RFC 7636 §4.1，64 字节随机 base64url）。
func SsoPKCEVerifier() string {
	b := make([]byte, 64)
	if _, err := rand.Read(b); err != nil {
		panic("crypto/rand failed: " + err.Error())
	}
	return base64.RawURLEncoding.EncodeToString(b)
}

// SsoPKCEChallenge S256: BASE64URL(SHA256(verifier))，无填充。
func SsoPKCEChallenge(verifier string) string {
	sum := sha256.Sum256([]byte(verifier))
	return base64.RawURLEncoding.EncodeToString(sum[:])
}

func ssoStateKey(state string) string  { return "sso:state:" + state }
func ssoTicketKey(t string) string     { return "sso:ticket:" + t }

func (s *ssoService) BuildAuthorizeURL(ctx context.Context, bindUID int) (string, error) {
	if !config.SsoEnabled() {
		return "", domain.ErrBadRequest
	}
	state := util.GenerateRandomString(32)
	verifier := SsoPKCEVerifier()
	payload, err := json.Marshal(domain.SsoStatePayload{Verifier: verifier, BindUID: bindUID})
	if err != nil {
		return "", err
	}
	if err := s.kv.Set(ctx, ssoStateKey(state), string(payload), ssoStateTTL); err != nil {
		return "", err
	}
	q := url.Values{}
	q.Set("response_type", "code")
	q.Set("client_id", config.SsoClientID())
	q.Set("redirect_uri", config.SsoRedirectURI())
	q.Set("scope", ssoScope)
	q.Set("state", state)
	q.Set("code_challenge", SsoPKCEChallenge(verifier))
	q.Set("code_challenge_method", "S256")
	return config.SsoAuthorizeURL() + "?" + q.Encode(), nil
}

// exchangeCode POST /api/oauth/token（grant_type=authorization_code + PKCE + client_secret）。
func (s *ssoService) exchangeCode(ctx context.Context, code, verifier string) (string, error) {
	form := url.Values{}
	form.Set("grant_type", "authorization_code")
	form.Set("code", code)
	form.Set("client_id", config.SsoClientID())
	form.Set("client_secret", config.SsoClientSecret())
	form.Set("code_verifier", verifier)
	form.Set("redirect_uri", config.SsoRedirectURI())
	req, err := http.NewRequestWithContext(ctx, http.MethodPost, config.SsoTokenURL(), strings.NewReader(form.Encode()))
	if err != nil {
		return "", err
	}
	req.Header.Set("Content-Type", "application/x-www-form-urlencoded")
	resp, err := s.http.Do(req)
	if err != nil {
		return "", err
	}
	defer resp.Body.Close()
	body, _ := io.ReadAll(io.LimitReader(resp.Body, 1<<20))
	if resp.StatusCode != http.StatusOK {
		return "", fmt.Errorf("sso token endpoint status %d: %s", resp.StatusCode, body)
	}
	var out struct {
		AccessToken string `json:"access_token"`
	}
	if err := json.Unmarshal(body, &out); err != nil || out.AccessToken == "" {
		return "", fmt.Errorf("sso token response invalid: %s", body)
	}
	return out.AccessToken, nil
}

// fetchUserinfo GET /api/oauth/userinfo，Bearer access_token。
func (s *ssoService) fetchUserinfo(ctx context.Context, accessToken string) (*domain.SsoUserinfo, error) {
	req, err := http.NewRequestWithContext(ctx, http.MethodGet, config.SsoUserinfoURL(), nil)
	if err != nil {
		return nil, err
	}
	req.Header.Set("Authorization", "Bearer "+accessToken)
	resp, err := s.http.Do(req)
	if err != nil {
		return nil, err
	}
	defer resp.Body.Close()
	body, _ := io.ReadAll(io.LimitReader(resp.Body, 1<<20))
	if resp.StatusCode != http.StatusOK {
		return nil, fmt.Errorf("sso userinfo status %d: %s", resp.StatusCode, body)
	}
	var info domain.SsoUserinfo
	if err := json.Unmarshal(body, &info); err != nil || info.Sub == "" {
		return nil, fmt.Errorf("sso userinfo invalid: %s", body)
	}
	return &info, nil
}

func (s *ssoService) ConsumeCallback(ctx context.Context, code, state string) (string, error) {
	if code == "" || state == "" {
		return "", ErrSsoStateInvalid
	}
	raw, err := s.kv.GetDel(ctx, ssoStateKey(state))
	if err != nil || raw == "" {
		return "", ErrSsoStateInvalid // 不存在/过期/重放
	}
	var st domain.SsoStatePayload
	if err := json.Unmarshal([]byte(raw), &st); err != nil || st.Verifier == "" {
		return "", ErrSsoStateInvalid
	}

	accessToken, err := s.exchangeCode(ctx, code, st.Verifier)
	if err != nil {
		return "", ErrSsoExchange
	}
	info, err := s.fetchUserinfo(ctx, accessToken)
	if err != nil {
		return "", ErrSsoExchange
	}

	// 分派：sub 已绑定 → 直接登录（或绑定场景下的幂等/冲突判定）
	binding, err := s.bindings.FindBySub(ctx, info.Sub)
	switch {
	case err == nil:
		if st.BindUID > 0 && st.BindUID != binding.LocalUserID {
			return "", ErrSsoConflict // 该 SSO 身份已绑在别的集市账号上
		}
		return s.saveTicket(ctx, &domain.SsoTicketPayload{Type: ssoTicketLogin, UID: binding.LocalUserID})
	case errors.Is(err, domain.ErrNotFound):
		// sub 未绑定，继续
	default:
		return "", err
	}

	if st.BindUID > 0 { // 已登录用户主动绑定：校验本账号未绑 → 直接写绑定
		if _, err := s.bindings.FindByUserID(ctx, st.BindUID); err == nil {
			return "", ErrSsoConflict // 本集市账号已有绑定
		}
		if err := s.bindings.Create(ctx, &domain.SsoBinding{
			LocalUserID: st.BindUID,
			SsoSub:      info.Sub,
			SsoUsername: info.PreferredUsername,
			CreatedAt:   time.Now(),
		}); err != nil {
			return "", err
		}
		return s.saveTicket(ctx, &domain.SsoTicketPayload{Type: ssoTicketBound, UID: st.BindUID})
	}

	// 未登录且未绑定 → 待绑定 ticket（前端展示绑定/注册引导）
	return s.saveTicket(ctx, &domain.SsoTicketPayload{
		Type:        ssoTicketBindNeeded,
		Sub:         info.Sub,
		SsoUsername: info.PreferredUsername,
		SsoEmail:    info.Email,
	})
}

func (s *ssoService) saveTicket(ctx context.Context, p *domain.SsoTicketPayload) (string, error) {
	ticket := util.GenerateRandomString(32)
	raw, err := json.Marshal(p)
	if err != nil {
		return "", err
	}
	if err := s.kv.Set(ctx, ssoTicketKey(ticket), string(raw), ssoTicketTTL); err != nil {
		return "", err
	}
	return ticket, nil
}

func (s *ssoService) ExchangeTicket(ctx context.Context, ticket string) (*domain.SsoExchangeResult, error) {
	p, err := s.peekTicket(ctx, ticket)
	if err != nil {
		return nil, err
	}
	switch p.Type {
	case ssoTicketLogin, ssoTicketBound:
		// 一次性：GETDEL 抢占，并发重放第二次落空
		if _, err := s.kv.GetDel(ctx, ssoTicketKey(ticket)); err != nil {
			return nil, domain.ErrUnauthorized
		}
		tokens, err := s.tokens.IssueTokensForUser(ctx, p.UID)
		if err != nil {
			return nil, err
		}
		return &domain.SsoExchangeResult{
			Type:         p.Type,
			Token:        tokens.AccessToken,
			RefreshToken: tokens.RefreshToken,
		}, nil
	case ssoTicketBindNeeded:
		// 不消费：等 POST /api/sso/bind 最终消费
		return &domain.SsoExchangeResult{
			Type:        p.Type,
			SsoUsername: p.SsoUsername,
			SsoEmail:    p.SsoEmail,
		}, nil
	}
	return nil, domain.ErrUnauthorized
}

func (s *ssoService) peekTicket(ctx context.Context, ticket string) (*domain.SsoTicketPayload, error) {
	if ticket == "" {
		return nil, domain.ErrUnauthorized
	}
	raw, err := s.kv.Get(ctx, ssoTicketKey(ticket))
	if err != nil || raw == "" {
		return nil, domain.ErrUnauthorized
	}
	var p domain.SsoTicketPayload
	if err := json.Unmarshal([]byte(raw), &p); err != nil || p.Type == "" {
		return nil, domain.ErrUnauthorized
	}
	return &p, nil
}

func (s *ssoService) consumeTicket(ctx context.Context, ticket string) (*domain.SsoTicketPayload, error) {
	if ticket == "" {
		return nil, domain.ErrUnauthorized
	}
	raw, err := s.kv.GetDel(ctx, ssoTicketKey(ticket))
	if err != nil || raw == "" {
		return nil, domain.ErrUnauthorized // 无效/过期/已消费
	}
	var p domain.SsoTicketPayload
	if err := json.Unmarshal([]byte(raw), &p); err != nil || p.Type == "" {
		return nil, domain.ErrUnauthorized
	}
	return &p, nil
}

func (s *ssoService) BindWithCredentials(ctx context.Context, ticket, email, password string) (*domain.AuthTokens, error) {
	email = strings.TrimSpace(strings.ToLower(email))
	if email == "" || len(password) < 6 {
		return nil, domain.ErrBadRequest
	}
	p, err := s.consumeTicket(ctx, ticket) // 一次性：无论后续成败，ticket 已消费
	if err != nil {
		return nil, err
	}
	if p.Type != ssoTicketBindNeeded || p.Sub == "" {
		return nil, domain.ErrUnauthorized
	}

	u, hash, err := s.users.FindAuthByEmail(ctx, email)
	if err != nil {
		if errors.Is(err, domain.ErrNotFound) {
			return nil, domain.ErrNotFound // 422：账号不存在
		}
		return nil, err
	}
	if bcrypt.CompareHashAndPassword([]byte(hash), []byte(password)) != nil {
		return nil, domain.ErrUnauthorized // 401：密码错误
	}
	if _, err := s.bindings.FindByUserID(ctx, u.ID); err == nil {
		return nil, domain.ErrConflict // 该集市账号已绑定其他 SSO
	}
	if err := s.bindings.Create(ctx, &domain.SsoBinding{
		LocalUserID: u.ID,
		SsoSub:      p.Sub,
		SsoUsername: p.SsoUsername,
		CreatedAt:   time.Now(),
	}); err != nil {
		return nil, err // 含并发下的唯一索引冲突 → ErrConflict
	}
	return s.tokens.IssueTokensForUser(ctx, u.ID)
}

func (s *ssoService) GetStatus(ctx context.Context, userID int) (*domain.SsoStatus, error) {
	b, err := s.bindings.FindByUserID(ctx, userID)
	if err != nil {
		if errors.Is(err, domain.ErrNotFound) {
			return &domain.SsoStatus{Bound: false}, nil
		}
		return nil, err
	}
	return &domain.SsoStatus{Bound: true, SsoUsername: b.SsoUsername, BoundAt: b.CreatedAt}, nil
}

func (s *ssoService) Unbind(ctx context.Context, userID int, password string) error {
	if userID <= 0 || len(password) < 6 {
		return domain.ErrBadRequest
	}
	u, err := s.users.FindByID(ctx, userID)
	if err != nil {
		return err
	}
	_, hash, err := s.users.FindAuthByEmail(ctx, u.Email)
	if err != nil {
		return err
	}
	if bcrypt.CompareHashAndPassword([]byte(hash), []byte(password)) != nil {
		return domain.ErrUnauthorized
	}
	return s.bindings.DeleteByUserID(ctx, userID)
}
```

**实现提醒**
- `FindAuthByEmail` 返回的 hash 与登录路径一致（bcrypt）。
- `domain.AuthTokens` 字段名以 `service/user_service.go` 为准（`AccessToken` / `RefreshToken`）。
- Redis `Get` 未命中返回 `redis.Nil` error —— 上面统一按 `err != nil || raw == ""` 处理，测试的内存 KV 未命中返回 `("", nil)` 时也能命中 `raw == ""` 分支。

### 6.1 新建 `controller/ssoController.go`

```go
package controller

import (
	"errors"
	"net/http"

	"loginTest/apperr"
	"loginTest/domain"
	"loginTest/response"
	"loginTest/service"
	"loginTest/util"

// SsoLogin GET /api/sso/login —— 未登录入口，302 到 UniSSO authorize。
func SsoLogin(c *gin.Context) {
	authorizeURL, err := service.Sso.BuildAuthorizeURL(c.Request.Context(), 0)
	if err != nil {
		response.Response(c, http.StatusServiceUnavailable, 503, nil, "SSO 登录未启用或暂时不可用")
		return
	}
	c.Redirect(http.StatusFound, authorizeURL)
}

// SsoBindStart POST /api/sso/bind/start —— 已登录用户发起绑定，前端拿 authorize_url 跳转。
func SsoBindStart(c *gin.Context) {
	user := authUser(c)
	if user == nil {
		response.Response(c, http.StatusUnauthorized, 401, nil, "用户未登录")
		return
	}
	if st, err := service.Sso.GetStatus(c.Request.Context(), user.ID); err == nil && st.Bound {
		response.Response(c, http.StatusConflict, 409, nil, "当前账号已绑定 SSO，请先解绑")
		return
	}
	authorizeURL, err := service.Sso.BuildAuthorizeURL(c.Request.Context(), user.ID)
	if err != nil {
		response.Response(c, http.StatusServiceUnavailable, 503, nil, "SSO 未启用或暂时不可用")
		return
	}
	response.Success(c, gin.H{"authorize_url": authorizeURL}, "获取绑定地址成功")
}

// SsoCallback GET /api/sso/callback —— UniSSO 回调，302 到前端结果页。
// 成功: /auth/sso/result?ticket=xxx；失败: /auth/sso/result?error=state|exchange|conflict
func SsoCallback(c *gin.Context) {
	ticket, err := service.Sso.ConsumeCallback(c.Request.Context(), c.Query("code"), c.Query("state"))
	if err != nil {
		errCode := "exchange"
		switch {
		case errors.Is(err, service.ErrSsoStateInvalid):
			errCode = "state"
		case errors.Is(err, service.ErrSsoConflict):
			errCode = "conflict"
		}
		c.Redirect(http.StatusFound, "/auth/sso/result?error="+errCode)
		return
	}
	c.Redirect(http.StatusFound, "/auth/sso/result?ticket="+ticket)
}

// SsoExchange POST /api/sso/exchange {ticket} —— 登录令牌或 bind_needed。
func SsoExchange(c *gin.Context) {
	var req struct {
		Ticket string `json:"ticket"`
	}
	if err := c.Bind(&req); err != nil || req.Ticket == "" {
		response.Response(c, http.StatusBadRequest, 400, nil, "请求参数错误")
		return
	}
	result, err := service.Sso.ExchangeTicket(c.Request.Context(), req.Ticket)
	if err != nil {
		response.Response(c, http.StatusUnauthorized, 401, nil, "登录凭证无效或已过期，请重新发起 SSO 登录")
		return
	}
	response.Success(c, result, "成功")
}

// SsoBind POST /api/sso/bind {ticket, email, password} —— 邮箱密码验证后写绑定并发令牌。
func SsoBind(c *gin.Context) {
	var req struct {
		Ticket   string `json:"ticket"`
		Email    string `json:"email"`
		Password string `json:"password"` // AES 加密传输
	}
	if err := c.Bind(&req); err != nil || req.Ticket == "" || req.Email == "" {
		response.Response(c, http.StatusBadRequest, 400, nil, "请求参数错误")
		return
	}
	tokens, err := service.Sso.BindWithCredentials(c.Request.Context(), req.Ticket, req.Email, util.Decrypt(req.Password))
	if err != nil {
		switch {
		case errors.Is(err, domain.ErrNotFound):
			response.Response(c, http.StatusUnprocessableEntity, 422, nil, "该邮箱未注册集市账号，请先注册")
		case errors.Is(err, domain.ErrUnauthorized):
			response.Response(c, http.StatusUnauthorized, 401, nil, "邮箱或密码错误")
		case errors.Is(err, domain.ErrConflict):
			response.Response(c, http.StatusConflict, 409, nil, "该账号或 SSO 已存在绑定")
		case errors.Is(err, domain.ErrBadRequest):
			response.Response(c, http.StatusBadRequest, 400, nil, "请求参数错误")
		default:
			apperr.WriteMsg(c, err, "绑定失败")
		}
		return
	}
	response.Success(c, gin.H{
		"token":         tokens.AccessToken,
		"refresh_token": tokens.RefreshToken,
	}, "绑定并登录成功")
}

// SsoStatus GET /api/sso/status —— 绑定状态。
func SsoStatus(c *gin.Context) {
	user := authUser(c)
	if user == nil {
		response.Response(c, http.StatusUnauthorized, 401, nil, "用户未登录")
		return
	}
	st, err := service.Sso.GetStatus(c.Request.Context(), user.ID)
	if err != nil {
		apperr.WriteMsg(c, err, "查询失败")
		return
	}
	response.Success(c, st, "成功")
}

// SsoUnbind POST /api/sso/unbind {password} —— 验证密码后解绑。
func SsoUnbind(c *gin.Context) {
	user := authUser(c)
	if user == nil {
		response.Response(c, http.StatusUnauthorized, 401, nil, "用户未登录")
		return
	}
	var req struct {
		Password string `json:"password"` // AES 加密传输
	}
	if err := c.Bind(&req); err != nil {
		response.Response(c, http.StatusBadRequest, 400, nil, "请求参数错误")
		return
	}
	if err := service.Sso.Unbind(c.Request.Context(), user.ID, util.Decrypt(req.Password)); err != nil {
		switch {
		case errors.Is(err, domain.ErrUnauthorized):
			response.Response(c, http.StatusUnauthorized, 401, nil, "密码错误")
		case errors.Is(err, domain.ErrNotFound):
			response.Response(c, http.StatusConflict, 409, nil, "当前账号未绑定 SSO")
		case errors.Is(err, domain.ErrBadRequest):
			response.Response(c, http.StatusBadRequest, 400, nil, "请求参数错误")
		default:
			apperr.WriteMsg(c, err, "解绑失败")
		}
		return
	}
	response.Success(c, nil, "解绑成功")
}
```

## Task 8 — 前端 API 封装 + 结果页（S6）

### 8.1 新建 `src/api/sso/sso.ts`

```ts
import { requestFunc } from '../req'
import { setPassword } from '../LoginAndRegister/utils'

const apiUrl = import.meta.env.VITE_API_BASE_URL

export interface SsoExchangeResult {
  type: 'login' | 'bind_needed' | 'bound'
  token?: string
  refresh_token?: string
  sso_username?: string
  sso_email?: string
}

export interface SsoStatus {
  bound: boolean
  sso_username?: string
  bound_at?: string
}

/** 未登录 SSO 入口：直接整页跳转后端 302 */
export function ssoLoginURL(): string {
  return `${apiUrl}/sso/login`
}

export async function ssoExchange(ticket: string): Promise<SsoExchangeResult | null> {
  const res = await requestFunc('/sso/exchange', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: { ticket },
  }, false)
  if (!res || !res.ok)
    return null
  const data = await res.json()
  return data?.data ?? null
}

export async function ssoBind(ticket: string, email: string, password: string): Promise<{ token: string, refresh_token: string } | null> {
  const res = await requestFunc('/sso/bind', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: { ticket, email, password: setPassword(password, '16bit secret key') },
  }, false)
  if (!res)
    return null
  const data = await res.json()
  if (!res.ok)
    throw new Error(typeof data?.msg === 'string' ? data.msg : '绑定失败')
  const token = data?.data?.token
  const refreshToken = data?.data?.refresh_token
  return token && refreshToken ? { token, refresh_token: refreshToken } : null
}

export async function ssoBindStart(): Promise<string | null> {
  const res = await requestFunc('/sso/bind/start', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: {},
  }, true)
  if (!res || !res.ok)
    return null
  const data = await res.json()
  return data?.data?.authorize_url ?? null
}

export async function ssoStatus(): Promise<SsoStatus | null> {
  const res = await requestFunc('/sso/status', {
    method: 'GET',
    headers: { 'Content-Type': 'application/json' },
  }, true)
  if (!res || !res.ok)
    return null
  const data = await res.json()
  return data?.data ?? null
}

export async function ssoUnbind(password: string): Promise<boolean> {
  const res = await requestFunc('/sso/unbind', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: { password: setPassword(password, '16bit secret key') },
  }, true)
  return !!res && res.ok
}
```

（`requestFunc` 签名/相对路径拼接以 `src/api/req.ts` 现状为准——若内部已拼 baseURL，则第一个参数照 `'/sso/exchange'` 写。）

```ts
import { ssoBindStart, ssoStatus, ssoUnbind } from '@/api/sso/sso'

const ssoOpen = ref(false)
const ssoBound = ref(false)
const ssoUsername = ref('')
const ssoBoundAt = ref('')

async function loadSsoStatus() {
  const st = await ssoStatus()
  ssoBound.value = !!st?.bound
  ssoUsername.value = st?.sso_username ?? ''
  ssoBoundAt.value = st?.bound_at ? new Date(st.bound_at).toLocaleDateString() : ''

function onSsoOpenChange(open: boolean) {
  if (open)
    loadSsoStatus()

async function startSsoBind() {
  const url = await ssoBindStart()

async function confirmUnbind() {

      showMsg('解绑成功')
      unbindPassword.value = ''
      await loadSsoStatus()
    }
  }
  finally {
    unbindBusy.value = false
  }
}
```

## 风险与回滚

- SSO 全部入口挂在 `SSO_ENABLED` 开关后：出问题置 `SSO_ENABLED=false`，登录页 SSO 按钮 503，其余系统零影响。

- client_secret 泄露处置：UniSSO 侧轮换 secret（服务方流程）+ 更新环境变量即可，集市侧无数据迁移。
